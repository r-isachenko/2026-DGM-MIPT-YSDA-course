\begin{frame}{Recap of Previous Lecture}
	\myfootnotewithlink{https://arxiv.org/abs/2107.03006}{Austin J. et al., Structured denoising diffusion models in discrete state-spaces, 2021.}
	\begin{block}{ELBO}
		\vspace{-0.7cm}
		\begin{multline*}
            \cL_{\bphi, \btheta}(\bx) =  {\color{olive}\bbE_{q(\bx_1 | \bx_0)} \log \pt(\bx_0 | \bx_1)} - {\color{violet}\KL\bigl(q(\bx_T | \bx_0) \| p(\bx_T)\bigr)} - \\
            - {\color{teal}\sum_{t=2}^T \underbrace{ \bbE_{q(\bx_t | \bx_0)} \KL \bigl(q(\bx_{t-1} | \bx_t, \bx_0) \| \pt(\bx_{t-1} | \bx_t)\bigr)}_{\cL_t}}
        \end{multline*}
		\vspace{-0.7cm}
	\end{block}
	\begin{block}{Discrete conditioned reverse distribution}
		\[
			q(\bx_{t-1} | \bx_t, \bx_0)
			= \Cat\left(\frac{\bQ_t \bx_t \odot \bQ_{1:t-1} \bx_0}{\bx_t^{\top}\bQ_{1:t}\bx_0}\right).
		\]
		\vspace{-0.3cm}
	\end{block}
	\begin{itemize}
		\item Both $q(\bx_{t-1} | \bx_t, \bx_0)$ and $q(\bx_t | \bx_0)$ are known analytically from the forward process.
		\item The reverse process $\pt(\bx_{t-1} | \bx_t)$ is a learned categorical distribution:
		\vspace{-0.3cm}
        \[
			\pt(\bx_{t-1} | \bx_t) = \Cat\bigl( \bpi_{\btheta}(\bx_t, t) \bigr).
		\]
	\end{itemize}
\end{frame}

\begin{frame}{Recap of Previous Lecture}
	\myfootnotewithlink{https://arxiv.org/abs/2107.03006}{Austin J. et al., Structured denoising diffusion models in discrete state-spaces, 2021.}

	\begin{block}{ELBO term}
		\vspace{-0.3cm}
		\[
			\cL_t
			= \bbE_{q(\bx_t | \bx_0)}
			\KL\bigl(q(\bx_{t-1} | \bx_t, \bx_0) \,\|\, \pt(\bx_{t-1} | \bx_t)\bigr).
		\]
		\vspace{-0.3cm}
	\end{block}
	\begin{block}{Categorical KL}
		\vspace{-0.3cm}
		\[
			\KL\bigl(\Cat(\bq) \,\|\, \Cat(\bp)\bigr)
			= \sum_{k=1}^K q_k \log \frac{q_k}{p_k}
			= \Ent(\bq,\bp) - \Ent(\bq),
		\]
		\begin{itemize}
			\item $\Ent\bigl(q(\bx_{t-1} | \bx_t, \bx_0)\bigr)$ is a constant w.r.t.~$\btheta$.
			\item $\Ent(\bq,\bp) = -\sum_k q_k \log p_k$ is a \textbf{cross-entropy loss}.
		\end{itemize}
	\end{block}
	Therefore, minimizing $\cL_t$ w.r.t.~$\btheta$ is equivalent to minimizing
	\[
		\bbE_{q(\bx_t | \bx_0)}
		\Ent\Bigl(q(\bx_{t-1} | \bx_t, \bx_0),\, \pt(\bx_{t-1} | \bx_t)\Bigr).
	\]
\end{frame}

\begin{frame}{Setup: from PF-ODE to few-step samplers}
	\myfootnotewithlink{https://arxiv.org/abs/2011.13456}{Song Y. et al. Score-Based Generative Modeling through Stochastic Differential Equations, 2020}
	\begin{block}{Recall: probability flow ODE}
		\vspace{-0.3cm}
		\[
			d\bx = \left[\bff(\bx, t) - \tfrac{1}{2} g^2(t)\, \bs(\bx, t)\right] dt, \quad \bs(\bx, t) = \nabla_{\bx} \log p_t(\bx).
		\]
	\end{block}
	\eqpause
	\begin{itemize}
		\item Plugging $\bs_{\btheta}$ into the PF-ODE and using a high-order \ODESolve (Heun, RK) reduces the number of steps from $100$--$1000$ down to $20$--$50$.
		\nextonslide{\item Can we go further --- to $\mathbf{1}$--$\mathbf{4}$ steps?}
		\nextonslide{\item Idea: \textbf{learn a shortcut network} that compresses many ODE-solver steps into one.}
	\end{itemize}
	\eqpause
	\begin{block}{Two approaches in this lecture}
		\begin{itemize}
			\item \textbf{Consistency Models} --- enforce self-consistency along PF-ODE trajectories.
			\item \textbf{Distribution Matching Distillation} --- match generator and data distributions via score differences.
		\end{itemize}
	\end{block}
\end{frame}

\section{Consistency Models}
%=======
\begin{frame}{Consistency function}
	\myfootnotewithlink{https://arxiv.org/abs/2303.01469}{Song Y. et al. Consistency Models, 2023}
	\begin{block}{Definition}
		A \emph{consistency function} $f_{\btheta}: (\bx_t, t) \mapsto \bx_{\epsilon}$ is a neural network that maps any noisy point on a PF-ODE trajectory back to its data-side endpoint $\bx_{\epsilon}$.
	\end{block}
	\eqpause
	\begin{figure}
		\centering
		\includegraphics[width=0.85\linewidth]{figs/pfode-fig1}
	\end{figure}
	\eqpause
	Domain $t \in [\epsilon, T]$: $\epsilon > 0$ (e.g.\ $0.002$) avoids the singularity at $t = 0$; $T$ is the noise level (e.g.\ $80$). Time scaling follows EDM: $\bx_t = \bx_0 + t \cdot \bepsilon$, $\bepsilon \sim \cN(0, \bI)$.
\end{frame}
%=======
\begin{frame}{Self-consistency property}
	\myfootnotewithlink{https://arxiv.org/abs/2303.01469}{Song Y. et al. Consistency Models, 2023}
	\begin{block}{Defining property}
		\vspace{-0.3cm}
		\[
			f_{\btheta}(\bx_t, t) = f_{\btheta}(\bx_{t'}, t'), \quad \forall t, t' \in [\epsilon, T],
		\]
		whenever $\bx_t$ and $\bx_{t'}$ lie on the same PF-ODE trajectory.
	\end{block}
	\eqpause
	\begin{figure}
		\centering
		\includegraphics[width=0.8\linewidth]{figs/cm-fig2}
	\end{figure}
	\eqpause
	The entire framework is built by enforcing this property along sampled trajectories.
\end{frame}
%=======
\begin{frame}{Boundary condition and parameterization}
	\myfootnotewithlink{https://arxiv.org/abs/2303.01469}{Song Y. et al. Consistency Models, 2023}
	\begin{block}{Boundary condition}
		\vspace{-0.3cm}
		\[
			f_{\btheta}(\bx_{\epsilon}, \epsilon) = \bx_{\epsilon}.
		\]
	\end{block}
	\eqpause
	\begin{block}{Skip-connection parameterization}
		\vspace{-0.3cm}
		\[
			f_{\btheta}(\bx, t) = c_{\text{skip}}(t)\, \bx + c_{\text{out}}(t)\, F_{\btheta}(\bx, t).
		\]
		\begin{itemize}
			\item $F_{\btheta}(\bx, t)$ --- free-form neural network (e.g.\ U-Net), the trainable predictor.
			\item $c_{\text{skip}}(t),\, c_{\text{out}}(t)$ --- differentiable scalar weights with $c_{\text{skip}}(\epsilon) = 1$ and $c_{\text{out}}(\epsilon) = 0$.
		\end{itemize}
	\end{block}
	\eqpause
	The boundary condition holds \emph{structurally} (by construction); the loss only has to enforce self-consistency.
\end{frame}
%=======
\begin{frame}{Consistency Distillation (CD)}
	\myfootnotewithlink{https://arxiv.org/abs/2303.01469}{Song Y. et al. Consistency Models, 2023}
	Assume a pretrained teacher score $\bs_{\bphi}(\bx, t)$. Discretize $[\epsilon, T]$ as $\epsilon = t_1 < t_2 < \dots < t_N = T$.
	\eqpause
	\begin{block}{One ODE-solver step on the empirical PF-ODE}
		\vspace{-0.3cm}
		\[
			\hat\bx_{t_n}^{\bphi} = \bx_{t_{n+1}} + (t_n - t_{n+1})\, \Phi(\bx_{t_{n+1}}, t_{n+1};\, \bphi).
		\]
		$\Phi(\bx, t; \bphi)$ --- update of a one-step \ODESolve\ for the empirical PF-ODE $d\bx/dt = -t \cdot \bs_{\bphi}(\bx, t)$ (e.g.\ Euler: $\Phi = -t\, \bs_{\bphi}$; Heun: average of two stage estimates).
	\end{block}
	\eqpause
	\begin{block}{CD loss}
		\vspace{-0.4cm}
		\[
			\cL_{\text{CD}}(\btheta) = \bbE \left[\lambda(t_n)\, d\bigl(f_{\btheta}(\bx_{t_{n+1}}, t_{n+1}),\; f_{\btheta^-}(\hat\bx_{t_n}^{\bphi}, t_n)\bigr)\right],
		\]
		with weighting $\lambda(t_n) > 0$; distance $d(\bu, \bv) \in \{\|\bu - \bv\|_2^2,\, \|\bu - \bv\|_1,\, \text{LPIPS}\}$; $\btheta^-$ --- EMA of $\btheta$ (stop-gradient target, prevents collapse to a constant).
	\end{block}
\end{frame}
%=======
\begin{frame}{Consistency Training (CT)}
	\myfootnotewithlink{https://arxiv.org/abs/2303.01469}{Song Y. et al. Consistency Models, 2023}
	Replace the teacher's ODE step with the noise-injection identity from DSM --- no pretrained score needed.
	\eqpause
	\begin{block}{Pair construction (teacher-free)}
		\vspace{-0.3cm}
		\[
			\bx_{t_{n+1}} = \bx_0 + t_{n+1}\, \bepsilon, \quad \bx_{t_n} = \bx_0 + t_n\, \bepsilon, \quad \bepsilon \sim \cN(0, \bI).
		\]
		The \emph{same} $\bx_0$ and the \emph{same} $\bepsilon$ for both points.
	\end{block}
	\eqpause
	\begin{block}{CT loss}
		\vspace{-0.4cm}
		\[
			\cL_{\text{CT}}(\btheta) = \bbE \left[\lambda(t_n)\, d\bigl(f_{\btheta}(\bx_{t_{n+1}}, t_{n+1}),\; f_{\btheta^-}(\bx_{t_n}, t_n)\bigr)\right].
		\]
	\end{block}
	\eqpause
	Claim: $\nabla_{\btheta}\, \cL_{\text{CT}} = \nabla_{\btheta}\, \cL_{\text{CD}} + o(\Delta t)$ as $N \to \infty$.
\end{frame}
%=======
\begin{frame}{Pseudocode (Consistency Training)}
	\myfootnotewithlink{https://arxiv.org/abs/2303.01469}{Song Y. et al. Consistency Models, 2023}
	\begin{block}{Training}
		\begin{enumerate}
			\item Sample $\bx_0 \sim \pd(\bx)$, $n \sim U\{1, N{-}1\}$, $\bepsilon \sim \cN(0, \bI)$.
			\item Compute paired noisy samples $\bx_{t_{n+1}} = \bx_0 + t_{n+1}\, \bepsilon$, $\bx_{t_n} = \bx_0 + t_n\, \bepsilon$.
			\item Compute loss $\cL = \lambda(t_n)\, d\bigl(f_{\btheta}(\bx_{t_{n+1}}, t_{n+1}),\; f_{\btheta^-}(\bx_{t_n}, t_n)\bigr)$.
			\item Update EMA target $\btheta^- \leftarrow \mu\, \btheta^- + (1 - \mu)\, \btheta$.
		\end{enumerate}
	\end{block}
	\eqpause
	\begin{block}{Sampling (Multi-step)}
		\begin{enumerate}
			\item Sample $\bx_T \sim \cN(0, T^2 \bI)$; denoise $\bx_0 \leftarrow f_{\btheta}(\bx_T, T)$.
			\item For $i = 1, \dots, M{-}1$: sample $\bepsilon \sim \cN(0, \bI)$, compute $\bx_{\tau_i} = \bx_0 + \tau_i\, \bepsilon$, denoise $\bx_0 \leftarrow f_{\btheta}(\bx_{\tau_i}, \tau_i)$.
			\item Return $\bx_0$.
		\end{enumerate}
	\end{block}
	\eqpause
	Multi-step trades a few extra NFEs ($M = 2$--$4$) for higher quality; $M = 1$ recovers single-step sampling.
\end{frame}
%=======
\begin{frame}{Latent Consistency Models (LCM)}
	\myfootnotewithlink{https://arxiv.org/abs/2310.04378}{Luo S. et al. Latent Consistency Models, 2023}
	Apply Consistency Distillation to a latent diffusion model (Stable Diffusion).
	\eqpause
	\begin{block}{Guided consistency distillation}
		Absorb classifier-free guidance into $f_{\btheta}$: an augmented function $f_{\btheta}(\bx_t, t, \omega, \bc)$ takes the guidance scale $\omega$ as input. The teacher uses CFG:
		\vspace{-0.2cm}
		\[
			\tilde\bs_{\bphi}(\bx_t, t, \omega, \bc) = (1 + \omega)\, \bs_{\bphi}(\bx_t, t, \bc) - \omega\, \bs_{\bphi}(\bx_t, t, \varnothing).
		\]
	\end{block}
	\eqpause
	\begin{block}{Skipping-step}
		Enforce consistency over $k$ steps instead of $1$:
		\vspace{-0.2cm}
		\[
			f_{\btheta}(\bx_{t_{n+k}}, t_{n+k}) \approx f_{\btheta^-}(\hat\bx_{t_n}, t_n).
		\]
		Larger SNR gap $\Rightarrow$ stronger training signal.
	\end{block}
\end{frame}

\section{Distribution Matching Distillation}
%=======
\begin{frame}{One-step generator}
	\myfootnotewithlink{https://arxiv.org/abs/2311.18828}{Yin T. et al. One-step Diffusion with Distribution Matching Distillation, 2023}
	\begin{block}{Setup}
		Train a generator $G_{\btheta}: \bz \mapsto \bx$, $\bz \sim \cN(0, \bI)$.
	\end{block}
	\eqpause
	\begin{block}{Objective}
		\vspace{-0.3cm}
		\[
			G_{\btheta} \# p(\bz) \approx \pd(\bx).
		\]
		\textbf{Distribution} matching, not per-sample matching.
	\end{block}
	\eqpause
	\begin{itemize}
		\item CM ties each $\bz$ to a specific PF-ODE endpoint.
		\item DMD lets $G_{\btheta}$ re-organize the noise-to-image map --- $\bz$ and $G_{\btheta}(\bz)$ need not lie on a teacher trajectory.
	\end{itemize}
\end{frame}
%=======
\begin{frame}{KL gradient via score difference}
	\myfootnotewithlink{https://arxiv.org/abs/2311.18828}{Yin T. et al. One-step Diffusion with Distribution Matching Distillation, 2023}
	Diffuse the generator output: $\bx_t = G_{\btheta}(\bz) + \sigma_t\, \bepsilon$. Let $p_{\text{fake}, t}$ and $p_{\text{real}, t}$ be the corresponding marginals.
	\eqpause
	\begin{block}{Variational score-distillation identity}
		\vspace{-0.6cm}
		\begin{multline*}
			\nabla_{\btheta}\, \KL\bigl(p_{\text{fake}, t} \,\|\, p_{\text{real}, t}\bigr) = \\
			= -\, \bbE \left[\bigl(\bs_{\text{real}, t}(\bx_t) - \bs_{\text{fake}, t}(\bx_t)\bigr) \cdot \nabla_{\btheta} G_{\btheta}(\bz) \right].
		\end{multline*}
		\vspace{-0.5cm}
	\end{block}
	\eqpause
	\begin{itemize}
		\item Likelihood-free: only score \emph{differences} matter.
		\item Integrate over $t$ with weighting $w(t)$ $\Rightarrow$ tractable training signal.
	\end{itemize}
\end{frame}
%=======
\begin{frame}{Two score networks}
	\myfootnotewithlink{https://arxiv.org/abs/2311.18828}{Yin T. et al. One-step Diffusion with Distribution Matching Distillation, 2023}
	\begin{block}{Real score (frozen)}
		$\bs_{\text{real}, t} = \bs_{\bphi}(\bx_t, t)$: the pretrained teacher.
	\end{block}
	\eqpause
	\begin{block}{Fake score (online)}
		$\bs_{\text{fake}, t} = \bs_{\bpsi}(\bx_t, t)$: a \emph{second} score network trained online with denoising score matching on samples from the current $G_{\btheta}$.
		\\
		Tracks $p_{\text{fake}}$ as the generator evolves.
	\end{block}
	\eqpause
	\begin{block}{Alternating optimization}
		\begin{enumerate}
			\item Update $\bs_{\bpsi}$ on a batch from $G_{\btheta}$ (score matching).
			\item Update $G_{\btheta}$ with the score-difference gradient.
		\end{enumerate}
	\end{block}
\end{frame}
%=======
\begin{frame}{DMD1: total loss}
	\myfootnotewithlink{https://arxiv.org/abs/2311.18828}{Yin T. et al. One-step Diffusion with Distribution Matching Distillation, 2023}
	\begin{block}{Total objective}
		\vspace{-0.3cm}
		\[
			\cL_{\text{DMD}}(\btheta) = \cL_{\text{KL-grad}}(\btheta) + \lambda_{\text{reg}}\, \cL_{\text{reg}}(\btheta).
		\]
	\end{block}
	\eqpause
	\begin{block}{Regression term}
		\vspace{-0.3cm}
		\[
			\cL_{\text{reg}}(\btheta) = \bbE_{(\bz, \bx^\star)} \| G_{\btheta}(\bz) - \bx^\star \|^2,
		\]
		on a few hundred precomputed $(\bz, \bx^\star)$ pairs ($\bx^\star$ = teacher's PF-ODE output for $\bz$).
	\end{block}
	\eqpause
	Interpretation: regression anchors $G_{\btheta}$ near the teacher's deterministic map; the KL-gradient term then re-shapes the \emph{distribution} of outputs.
\end{frame}
%=======
\begin{frame}{DMD2: improvements}
	\myfootnotewithlink{https://arxiv.org/abs/2405.14867}{Yin T. et al. Improved Distribution Matching Distillation for Fast Image Synthesis, 2024}
	Three changes over DMD1:
	\eqpause
	\begin{itemize}
		\item \textbf{GAN loss}: a discriminator on real images corrects the bias from an imperfect $\bs_{\bpsi}$.
		\nextonslide{\item \textbf{No regression}: removes the need for precomputed $(\bz, \bx^\star)$ pairs; simpler pipeline.}
		\nextonslide{\item \textbf{Multi-step generator}: $G_{\btheta}$ becomes a $4$-step network, recovering quality lost at the strict $1$-step setting.}
	\end{itemize}
	\eqpause
	Closes most of the quality gap between distilled and full-step diffusion.
\end{frame}
%=======
\begin{frame}{Summary}
	\begin{itemize}
		\item PF-ODE solvers (Heun, RK) floor at ${\sim}20$ steps; further acceleration requires \textbf{learning a shortcut network}.
		\item \textbf{Consistency Models}: enforce \emph{self-consistency} along PF-ODE trajectories; the boundary condition makes one-step sampling fall out of the parameterization.
		\item LCM extends CD to latent SD with guided distillation and skipping-step.
		\item \textbf{DMD}: match generator and data \emph{distributions} via the score-difference gradient; two score networks (frozen real, online fake) in alternating optimization. DMD2 adds GAN, drops regression, adds multi-step.
		\item Tradeoffs: CM is more principled, DMD currently delivers the best $1$-step quality.
	\end{itemize}
\end{frame}
